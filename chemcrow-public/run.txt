python - <<'PY'
import json
import time
from chemcrow.tools import EnzymeRetrosynthesisSearch

tool = EnzymeRetrosynthesisSearch(
    config={
        "use_database": True,
        "use_model": True,
        "max_depth": 3,
        "max_expansions": 50,
        "forward_anchor_max_nodes": 200,
        "max_model_calls": 1,
    }
)

payload = {
    "target": "CCO",
    "building_blocks": [
        "OC1OC[C@H](O)[C@H](O)[C@H]1O"
    ],
    "top_k": 5,
    "search_algorithm": "best_first_beam",
}

print("开始搜索……", flush=True)
started = time.time()

result = tool.run(json.dumps(payload))

print(f"搜索结束，耗时 {time.time() - started:.1f} 秒", flush=True)

# 工具返回的是 JSON 字符串
data = json.loads(result)
print(json.dumps(data, ensure_ascii=False, indent=2))

python - <<'PY'
import os
from chemcrow.agents import ChemCrow

TARGET = "CCO"
SUBSTRATE = "OC1OC[C@H](O)[C@H](O)[C@H]1O"

agent = ChemCrow(
    model="deepseek-flash",
    tools_model="deepseek-flash",
    openai_api_key=os.environ["DEEPSEEK_API_KEY"],
    api_base="https://api.deepseek.com",
    temp=0.0,
    streaming=False,
    max_iterations=40,
)

print("模型：", agent.llm.model_name)
print("API地址：", agent.llm.openai_api_base)

question = f"""
目标物：{TARGET}
可用底物：{SUBSTRATE}

不必调用安全检查工具。

请先规范化目标物和底物，然后使用 EnzymeRetrosynthesisSearch，
通过 AND-OR MCTS 搜索从给定底物到目标物的完整酶催化逆合成路径。

输出 Top 5 路线，格式为：
目标物 ←[EC编号；反应名称；反应ID；来源] 中间体 ←[...] 底物。

如果反应包含多个必要底物，请同时列出完整的 AND 反应方程。
如果没有找到完整路线，请明确报告搜索终止原因和已扩展节点数。
"""

result = agent.run(question)
print("\n最终结果：")
print(result)
